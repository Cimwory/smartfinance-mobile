<script>
        // Password Visibility Toggle
        const toggleBtn = document.getElementById('togglePassword');
        const passwordInput = document.getElementById('password');
        const visibilityIcon = document.getElementById('visibilityIcon');

        toggleBtn.addEventListener('click', () => {
            if (passwordInput.type === 'password') {
                passwordInput.type = 'text';
                visibilityIcon.textContent = 'visibility';
            } else {
                passwordInput.type = 'password';
                visibilityIcon.textContent = 'visibility_off';
            }
        });

        // Password Strength Logic
        const liquidBar = document.getElementById('liquidBar');
        const strengthText = document.getElementById('strengthText');

        passwordInput.addEventListener('input', (e) => {
            const val = e.target.value;
            let strength = 0;
            
            if (val.length > 0) strength = 1; // Weak
            if (val.length >= 8 && /[A-Z]/.test(val) && /[0-9]/.test(val)) strength = 2; // Medium
            if (val.length >= 10 && /[^A-Za-z0-9]/.test(val) && strength === 2) strength = 3; // Strong

            if (strength === 0) {
                liquidBar.style.width = '0%';
                strengthText.textContent = '';
            } else if (strength === 1) {
                liquidBar.style.width = '30%';
                strengthText.textContent = 'Weak';
                strengthText.style.color = '#ffb4ab';
            } else if (strength === 2) {
                liquidBar.style.width = '60%';
                strengthText.textContent = 'Medium';
                strengthText.style.color = '#FBBC05';
            } else if (strength === 3) {
                liquidBar.style.width = '90%';
                strengthText.textContent = 'Secure';
                strengthText.style.color = '#00e5ff';
            }
        });

        // Form submission prevent default for demo
        document.getElementById('registerForm').addEventListener('submit', (e) => {
            e.preventDefault();
            // Handle registration logic
            const btn = e.target.querySelector('button[type="submit"]');
            btn.innerHTML = '<span class="material-symbols-outlined animate-spin">sync</span> Processing...';
            setTimeout(() => {
                btn.innerHTML = 'Account Created <span class="material-symbols-outlined">check_circle</span>';
            }, 1500);
        });
    </script>
